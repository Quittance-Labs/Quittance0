# Server Configuration
PORT=3001
NODE_ENV=development

# --- Storage Backend (Issue #452) ---
# Controls which invoice repository the server uses.
#
#   INVOICE_STORAGE=memory   → in-memory MVP (default; no DATABASE_URL needed)
#   INVOICE_STORAGE=postgres → PostgreSQL (requires DATABASE_URL below)
#   (unset)                  → postgres when DATABASE_URL is present, else memory
#
# If INVOICE_STORAGE=postgres is set but DATABASE_URL is absent, the server
# exits immediately at boot with an explicit error.
#
# Memory is the safe local-demo default.  Set postgres only when you have
# provisioned a database and run npm run db:migrate.
INVOICE_STORAGE=memory

# Database
# db/schema.sql defines the invoices table with full parity columns:
#   seller_name, seller_email, asset_code, asset_issuer, customer_name,
#   customer_email, payer_public_key, payer_name, payer_email,
#   expires_at, paid_at, metadata (JSONB).
#
# Both server.ts (Postgres) and server-mvp.ts (in-memory) expose the same
# invoice shape through the shared InvoiceStorage interface.
DATABASE_URL=postgresql://user:password@localhost:5432/stellar_invoices

# Stellar Network (Validated by /api/ready)
# Readiness fails fast with 503 if STELLAR_NETWORK is not TESTNET or PUBLIC,
# or if STELLAR_HORIZON_URL does not use HTTPS in production.
STELLAR_NETWORK=TESTNET
# TESTNET or PUBLIC
STELLAR_HORIZON_URL=https://horizon-testnet.stellar.org

# Optional: Enable lightweight Horizon reachability check on readiness probes.
# Default: false (off for demo deploys so external latency does not trigger false negatives)
